Practice: Disks, Filesystems and Storage Management



Partitioning (MBR and GPT)

MBR Partitioning (fdisk, 512 MBR Table):

	lsblk - (sda, sdb, sdc - three drivers)

	sudo fdisk /dev/sdb (create two partitions on drive sdb)
		- p -> default -> +200M (creates a 200 MB partition)
		- p -> default -> +300M (creates a 300 MB partition)
		- w (to write the changes and exit the fdisk menu)
	sudo dd if=/dev/sdb of=512.mbr bs=512 count=1 (create a backup of the MBR table)

	sudo dd if=/dev/zero of=/dev/sdb bs=512 count=1 (manually destroying the MBR table - all paritions disappear after this, because MBR table is overwritten with zeros)
	sudo dd if=~/512.mbr of=/dev/sdb bs=512 count=1 (recomver the MBR table from the backup created)

	sudo lsblk -l /dev/sdb  (to confirm that everything is fixed)



GPT Partitioning (gdisk, 17K GPT Table):

	sudo dnf install -y gdisk (might not be installed by default)

	sudo dd if=/dev/zero of=/dev/sdb bs=512 count=1 (format the drive before we start using GPT)

	sudo gdisk /dev/sdb
		- n -> 1 -> default -> +200M -> 8300 (for Linux filesystem)
		- n -> 2 -> default -> +300M -> 8300
		- w -> y

	lsblk -l /dev/sdb (to confirm the changes)


	sudo dd if=/dev/zero of=/dev/sdb bs=17K count=1 (there is a backup GPT table, so we will test the recovery feature)
	sudo gdisk /dev/sdb (Partition table scan: GPT: damaged)
	- 1 -> p -> w -> y (save changes and exit, done)

	lsblk (partitions are recovered)



Filesystems (ext4 and xfs):

	sudo mkdir -p /disks/{ext4,xfs}


EXT4:
	sudo mkfs.ext4 -L EXT4-VOLUME /dev/sdb1
	sudo mount /dev/sdb1 /disks/ext4

XFS:
	sudo mkfs.xfs -L XFS-VOLUME /dev/sdb2
	sudo mount /dev/sdb2 /disks/xfs


Permanent Mounting (fstab, /etc/fstab - mount the file system each time the machine is booted):
	sudo blkid /dev/sdb*  (we take the UUID value)
	sudo vi /etc/fstab    (add the two UUIDs of the file systems)
	
	sudo mount -a
	mount | grep sdb   or cat /proc/mounts | grep sdb    (check if everything is correctly added)



Logical Volume Management (LVM):
	sudo gdisk /dev/sdb  (create 1 more partition on the second drive)
	- n -> 3 -> default -> +300M -> 8300
	- w -> y

	sudo fdisk /dev/sdc (create another partition on the third drive)
	- n -> 1 -> default -> +400M
	- t -> 1 -> 44 (for Linux LVM)
	- w

	sudo partprobe -s
	lsblk

	sudo pvcreate /dev/sdb1  (creates a lv)
	sudo pvcreate /dev/sdc3
	sudo pvscan 

	sudo vgcreate vg1 /dev/sdb1 /dev/sdc3  (create a logic volume group)
	sudo vgscan  (should show vg1)

	sudo lvcreate -n lv1 -L 600m vg1 (now create a lv named lv1)
	sudo lvscan

	sudo mkfs.xfs -L LVM-XFS /dev/vg1/lv1
	sudo mkdir -p /disks/lvm
	sudo mount /dev/vg1/lv1 /disks/lvm

	sudo lvextend -l +100%FREE /dev/vg1/lv1 (first we extend the logic volume)
	sudo xfs_growfs /disks/lvm
	lsblk (to confirm that it is extended using 100% of the free space, 600M -> 692M)

	sudo blkid /dev/vg1/lv1  or /dev/mapper/vg1-lv1 (this will show us the UUID, so we can add it permanently in /etc/fstab)
	mount -a
	mount | grep lvm

Software RAID:
	sudo dnf install -y mdadm

	sudo fdisk /dev/sdb
	- n -> default -> default -> default
	- t - fd
	- p (to confirm the changes)
	- w

	We should do the fdisk steps for the other drives or execute this 3rd party tool command:
	sudo parted -s /dev/sdb -- mklabel msdos mkpart primary 1m -0m set 1 raid on
	sudo parted -s /dev/sdc -- mklabel msdos mkpart primary 1m -0m set 1 raid on
	sudo parted -s /dev/sdd -- mklabel msdos mkpart primary 1m -0m set 1 raid on
	sudo parted -s /dev/sde -- mklabel msdos mkpart primary 1m -0m set 1 raid on
	lsblk (to confirm the changes)

	sudo mdadm --create /dev/md0 --level=0 --raid-devices=2 /dev/sdb1 /dev/sdc1   (create Raid0 array out of the first two drivers)
	cat /proc/mdstat  or sudo mdadm -D /dev/md0  (check the RAID0 status)

	sudo mdadm --create /dev/md1 --level=1 --raid-devices=2 /dev/sdd1 /dev/sde1
	cat /proc/mdstat or sudo mdadm -D /dev/md1 (check the RAID1 status)

	sudo mdadm --detail --scan | sudo tee -a /etc/mdadm.conf (so the RAID can be reassembled on boot)

	sudo mkfs.ext4 /dev/md0 ; sudo mkfs.ext4 /dev/md1
	sudo mkdir -p /disks/raid{0,1}
	sudo mount /dev/md0 /disks/raid0/
	sudo mount /dev/md1 /disks/raid1/ 
	
	echo '/dev/md0 /disks/raid0 ext4 defaults 0 0' | sudo tee -a /etc/fstab
	echo '/dev/md1 /disks/raid1 ext4 default 0 0' | sudo tee -a /etc/fstab
